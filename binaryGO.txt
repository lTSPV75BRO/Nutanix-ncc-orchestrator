#!/usr/bin/env bash
# Release build for ncc-orchestrator, ncc-api-server, and ncc-ui-server.
#
# Hosts (same script, same flags):
#   Linux / macOS:   bash binaryGO.txt
#   Windows:         binaryGO.cmd
#                    or, from Git Bash: bash binaryGO.txt
#
# Default output is all six platform pairs:
#   linux, darwin, windows  ×  amd64, arm64
# Cross-compilation uses CGO_ENABLED=0, so a Linux or Windows host can
# produce the Mac and Windows binaries too. Each file is checked for an
# ELF, Mach-O, or PE header before the run is called successful.
#
# Examples:
#   bash binaryGO.txt
#   bash binaryGO.txt --os linux,darwin,windows --arch amd64,arm64 --jobs 8
#   bash binaryGO.txt --bins-only --skip-tests
#   bash binaryGO.txt --set-version 2.2.0
#   bash binaryGO.txt --update-deps --govulncheck
#
# Link flags always include main.Version, main.Stream, main.GitRevision,
# main.BuildDate, and main.GoVersion. Optional --docker packages the Linux
# binaries after they exist; it is not required to produce release binaries.
set -euo pipefail

uname_s=$(uname -s 2>/dev/null || echo unknown)
case "$uname_s" in
  Linux*) HOST_KERNEL=linux ;;
  Darwin*) HOST_KERNEL=darwin ;;
  MINGW*|MSYS*|CYGWIN*) HOST_KERNEL=windows ;;
  *)
    if [ "${OS:-}" = "Windows_NT" ]; then
      HOST_KERNEL=windows
    else
      HOST_KERNEL=unknown
    fi
    ;;
esac
# Git Bash rewrites Unix-looking arguments before they reach go.exe.
# Leave relative dist/ paths alone so -o and -ldflags survive.
if [ "$HOST_KERNEL" = "windows" ]; then
  export MSYS_NO_PATHCONV=1
  export MSYS2_ARG_CONV_EXCL='*'
fi

START_EPOCH=$(date +%s)

SKIP_TESTS=0
SKIP_PREFLIGHT=0
SKIP_FRONTEND=0
SKIP_VERSION_CHECK=0
SKIP_SBOM=0
NO_STACKS=0
BINS_ONLY=0
DO_DOCKER=0
DO_PUSH=0
DO_LATEST=1
UPDATE_DEPS=0
DO_GOVULN=0
REQUIRE_CLEAN=0
WITH_MCP=0
DRY_RUN=0
PRINT_LDFLAGS=0
SET_VERSION=""
VERSION_ARG=""
STREAM_ARG=""
OS_LIST="linux,darwin,windows"
ARCH_LIST="amd64,arm64"
DOCKER_ARCH=""
ENGINE_ARG=""
JOBS=""
REGISTRY="${NCC_DOCKER_REGISTRY:-docker.io}"
IMAGE_PREFIX="${NCC_IMAGE_PREFIX:-prajwalnutant}"

Version and dependencies

What to build

  --jobs N               Parallel go builds (default: CPU count, max 8)

die() {
  echo "binaryGO.txt: $*" >&2
  exit 2
}

in_csv() {
  case ",$1," in
    *",$2,"*) return 0 ;;
    *) return 1 ;;
  esac
}

csv_count() {
  local IFS=,
  set -- $1
  echo $#
}

run_python() {
  if command -v python3 >/dev/null 2>&1; then
    python3 "$@"
  elif command -v python >/dev/null 2>&1; then
    python "$@"
  elif command -v py >/dev/null 2>&1; then
    py -3 "$@"
  else
    return 127
  fi
}

run_npm() {
  if command -v npm >/dev/null 2>&1; then
    npm "$@"
  elif command -v npm.cmd >/dev/null 2>&1 && command -v cmd.exe >/dev/null 2>&1; then
    cmd.exe /c npm.cmd "$@"
  else
    echo "npm was not found on PATH" >&2
    return 127
  fi
}

default_jobs() {
  local n=4
  if [ "$HOST_KERNEL" = "windows" ] && [ -n "${NUMBER_OF_PROCESSORS:-}" ]; then
    n=$NUMBER_OF_PROCESSORS
  elif command -v nproc >/dev/null 2>&1; then
    n=$(nproc)
  elif command -v sysctl >/dev/null 2>&1; then
    n=$(sysctl -n hw.ncpu 2>/dev/null || echo 4)
  fi
  if [ "$n" -gt 8 ]; then n=8; fi
  if [ "$n" -lt 1 ]; then n=1; fi
  echo "$n"
}

sha256_file() {
  local sum win
  if command -v sha256sum >/dev/null 2>&1; then
    sha256sum "$1" | awk '{print $1}'
    return
  fi
  if command -v shasum >/dev/null 2>&1; then
    shasum -a 256 "$1" | awk '{print $1}'
    return
  fi
  if command -v openssl >/dev/null 2>&1; then
    openssl dgst -sha256 "$1" | awk '{print $NF}'
    return
  fi
  if command -v powershell.exe >/dev/null 2>&1; then
    win="$1"
    if command -v cygpath >/dev/null 2>&1; then
      win=$(cygpath -w "$1")
    fi
    sum=$(powershell.exe -NoProfile -Command "(Get-FileHash -Algorithm SHA256 -LiteralPath '$win').Hash.ToLower()")
    printf '%s\n' "$sum" | tr -d '\r'
    return
  fi
  echo "binaryGO.txt: no sha256 tool (sha256sum, shasum, openssl, or powershell)" >&2
  return 1
}

# First 4 bytes decide ELF / Mach-O / PE so a cross-compile that emitted the
# wrong container fails before archives are packed.
assert_binary_kind() {
  local file="$1" goos="$2" mag
  if ! command -v od >/dev/null 2>&1; then
    return 0
  fi
  mag=$(od -An -t x1 -N 4 "$file" | tr -d ' \n' | tr 'A-F' 'a-f')
  case "$goos" in
    linux)
      case "$mag" in
        7f454c46*) return 0 ;;
      esac
      ;;
    windows)
      case "$mag" in
        4d5a*) return 0 ;;
      esac
      ;;
    darwin)
      case "$mag" in
        cffaedfe*|feedfacf*|cefaedfe*|feedface*|cafebabe*|bebafeca*) return 0 ;;
      esac
      ;;
  esac
  echo "binaryGO.txt: $file is not a $goos binary (header $mag)" >&2
  return 1
}

link_or_copy() {
  local src="$1" dst="$2"
  rm -f "$dst"
  # Symlink target is the sibling filename so the link stays valid inside dist/.
  # Windows copies instead: creating symlinks there needs extra privileges.
  if [ "$HOST_KERNEL" != "windows" ] && ln -sfn "$(basename "$src")" "$dst" 2>/dev/null; then
    return 0
  fi
  cp -f "$src" "$dst"
}

pack_zip() {
  local root="$1" dest="$2"
  shift 2
  if command -v zip >/dev/null 2>&1; then
    (cd "$root" && zip -qr "$dest" "$@")
    return
  fi
  if run_python -c 'import zipfile' >/dev/null 2>&1; then
    (
      cd "$root" && run_python - "$dest" "$@" <<'PY'
import os, sys, zipfile
dest, items = sys.argv[1], sys.argv[2:]
with zipfile.ZipFile(dest, "w", compression=zipfile.ZIP_DEFLATED) as zf:
    for item in items:
        if os.path.isfile(item):
            zf.write(item, item.replace("\\", "/"))
            continue
        for dirpath, _, files in os.walk(item):
            for name in files:
                full = os.path.join(dirpath, name)
                zf.write(full, full.replace("\\", "/"))
PY
    )
    return
  fi
  die "Windows archives need zip or python (zipfile)"
}

count_matches() {
  local n=0 f pattern="$1"
  shopt -s nullglob
  for f in $pattern; do
    case "$f" in
      *.asc|*.sig) continue ;;
    esac
    if [ -f "$f" ]; then
      n=$((n + 1))
    fi
  done
  shopt -u nullglob
  echo "$n"
}

  if [ $# -lt 2 ] || [ -z "${2:-}" ]; then
    die "$1 needs a value"
  fi
}

stamp_version() {
  local ver="$1"
  run_python - "$ver" <<'PY'
import pathlib, re, sys
ver = sys.argv[1]
root = pathlib.Path(".")
(root / "VERSION").write_text(ver + "\n")

pkg = root / "frontend" / "package.json"
text = pkg.read_text()
pkg.write_text(re.sub(r'("version": ")[^"]+"', r"\g<1>" + ver + '"', text, count=1))

chart = root / "helm" / "ncc-orchestrator" / "Chart.yaml"
ct = chart.read_text()
ct = re.sub(r'(?m)^version:\s*\S+', "version: " + ver, ct, count=1)
ct = re.sub(r'(?m)^appVersion:\s*"[^"]+"', 'appVersion: "' + ver + '"', ct, count=1)
chart.write_text(ct)

values = root / "helm" / "ncc-orchestrator" / "values.yaml"
vt = values.read_text()
vt = re.sub(r'(?m)^(\s*tag:\s*")[^"]+"', r"\g<1>" + ver + '"', vt)
values.write_text(vt)

kust = root / "k8s" / "kustomization.yaml"
kt = kust.read_text()
kt = re.sub(r'(?m)^(\s*newTag:\s*")[^"]+"', r"\g<1>" + ver + '"', kt)
kust.write_text(kt)

img = re.compile(r'(docker\.io/prajwalnutant/(?:nutanix-ncc-orchestrator|ncc-api-server|ncc-ui-server):)[^\s"]+')
for path in (root / "k8s").glob("*.yaml"):
    raw = path.read_text()
    updated = img.sub(r"\g<1>" + ver, raw)
    if updated != raw:
        path.write_text(updated)
print("stamped version", ver)
PY
}

check_versions() {
  local failed=0 got tags
  got=$(awk -F'"' '/"version":/{print $4; exit}' frontend/package.json)
  if [ "$got" != "$VERSION" ]; then
    echo "version mismatch: frontend/package.json is $got, VERSION is $VERSION" >&2
    failed=1
  fi
  got=$(awk '/^version:/{print $2; exit}' helm/ncc-orchestrator/Chart.yaml)
  if [ "$got" != "$VERSION" ]; then
    echo "version mismatch: helm chart version is $got, VERSION is $VERSION" >&2
    failed=1
  fi
  got=$(awk -F'"' '/^appVersion:/{print $2; exit}' helm/ncc-orchestrator/Chart.yaml)
  if [ "$got" != "$VERSION" ]; then
    echo "version mismatch: helm appVersion is $got, VERSION is $VERSION" >&2
    failed=1
  fi
  tags=$(awk -F'"' '/newTag:/{print $2}' k8s/kustomization.yaml)
  for got in $tags; do
    if [ "$got" != "$VERSION" ]; then
      echo "version mismatch: k8s/kustomization.yaml newTag $got, VERSION is $VERSION" >&2
      failed=1
    fi
  done
  tags=$(awk -F'"' '/^[[:space:]]*tag:/{print $2}' helm/ncc-orchestrator/values.yaml)
  for got in $tags; do
    if [ "$got" != "$VERSION" ]; then
      echo "version mismatch: helm values tag $got, VERSION is $VERSION" >&2
      failed=1
    fi
  done
  local line tag
  for line in $(grep -h -E 'image: docker.io/prajwalnutant/.+:' k8s/*.yaml | awk '{print $2}'); do
    tag="${line##*:}"
    if [ "$tag" != "$VERSION" ]; then
      echo "version mismatch: k8s image $line, VERSION is $VERSION" >&2
      failed=1
    fi
  done
  if [ "$failed" -ne 0 ]; then
    echo "fix the stamps or re-run with --set-version $VERSION (or --skip-version-check)" >&2
    return 1
  fi
  echo "version stamps match $VERSION"
}

update_deps() {
  echo "updating Go modules..."
  go get -u ./...
  go mod tidy
  echo "updating frontend npm packages..."
  (cd frontend && run_npm update && run_npm audit --omit=dev)
  if ! command -v govulncheck >/dev/null 2>&1; then
    go install golang.org/x/vuln/cmd/govulncheck@latest
    export PATH="$(go env GOPATH)/bin:${PATH}"
  fi
  govulncheck ./...
  echo "dependency update finished; review git diff before committing"
}

resolve_container_engine() {
  local want="${ENGINE_ARG:-${NCC_CONTAINER_ENGINE:-}}"
  if [ -n "$want" ]; then
    case "$want" in
      docker|podman) ;;
      *) die "--engine must be docker or podman" ;;
    esac
    command -v "$want" >/dev/null 2>&1 || die "$want was not found on PATH"
    ENGINE="$want"
    return
  fi
  if command -v docker >/dev/null 2>&1; then
    ENGINE=docker
  elif command -v podman >/dev/null 2>&1; then
    ENGINE=podman
  else
    die "docker or podman is required for --docker. A shell alias such as alias docker=podman is not visible to this script; pass --engine podman"
  fi
}

  if [ -n "${FRONTEND_PID:-}" ]; then
    kill "$FRONTEND_PID" >/dev/null 2>&1 || true
    wait "$FRONTEND_PID" >/dev/null 2>&1 || true

      tail -200 "$testlog" >&2
      exit 1
    fi
    echo "go test ok"
  fi

fi

    go run ./tools/winversioninfo --version="$VERSION" --git-rev="$GIT_REV_SHORT" --output-dir=. --arch="$arch"
  done
fi

build_one() {
  local goos="$1" goarch="$2" pkg="$3" name ext flags
  ext=""
  if [ "$goos" = "windows" ]; then
    ext=".exe"
  fi
  if [ "$pkg" = "." ]; then
    name="ncc-orchestrator"
  else
    name=$(basename "$pkg")
  fi
  flags="$LDFLAGS"
  if [ "$name" = "ncc-mcp-server" ]; then
    flags="$flags -X main.serverVersion=${VERSION}${GIT_DIRTY}"
  fi
  echo "build ${name}-${goos}-${goarch}${ext}"
  CGO_ENABLED=0 GOOS="$goos" GOARCH="$goarch" go build -trimpath -buildvcs=true -ldflags "$flags" -o "dist/${name}-${goos}-${goarch}${ext}" "$pkg"
}

wait_batch() {
  local pid
  for pid in "$@"; do
    wait "$pid"
  done
}

pids=()
for goos in $(echo "$OS_LIST" | tr ',' ' '); do

verify_targets() {
  local goos goarch name ext path bad=0
  for goos in $(echo "$OS_LIST" | tr ',' ' '); do
    for goarch in $(echo "$ARCH_LIST" | tr ',' ' '); do
      ext=""
      if [ "$goos" = "windows" ]; then
        ext=".exe"
      fi
      names="ncc-orchestrator ncc-api-server ncc-ui-server"
      if [ "$WITH_MCP" = 1 ]; then
        names="${names} ncc-mcp-server"
      fi
      for name in $names; do
        path="dist/${name}-${goos}-${goarch}${ext}"
        if [ ! -f "$path" ]; then
          echo "missing ${path}" >&2
          bad=1
          continue
        fi
        if ! assert_binary_kind "$path" "$goos"; then
          bad=1
          continue
        fi
        echo "ok ${goos}/${goarch} ${name}"
      done
    done
  done
  if [ "$bad" -ne 0 ]; then
    echo "RELEASE_INCOMPLETE — one or more linux/mac/windows binaries failed the format check" >&2
    return 1
  fi
}

sign_binaries() {
  if [ -n "${NCC_MACOS_SIGN_IDENTITY:-}" ] && command -v codesign >/dev/null 2>&1 && in_csv "$OS_LIST" darwin; then
    local f
    shopt -s nullglob
    for f in dist/ncc-orchestrator-darwin-* dist/ncc-api-server-darwin-* dist/ncc-ui-server-darwin-*; do
      codesign --force --options runtime --timestamp --sign "$NCC_MACOS_SIGN_IDENTITY" "$f"
      codesign --verify --strict --verbose=2 "$f"
    done
    if [ -n "${NCC_MACOS_NOTARY_PROFILE:-}" ] && command -v xcrun >/dev/null 2>&1; then
      for f in dist/ncc-orchestrator-darwin-* dist/ncc-api-server-darwin-* dist/ncc-ui-server-darwin-*; do
        ditto -c -k --keepParent "$f" "$f.zip"
        xcrun notarytool submit "$f.zip" --keychain-profile "$NCC_MACOS_NOTARY_PROFILE" --wait
        rm -f "$f.zip"
      done
    fi
    shopt -u nullglob
  fi
  if [ -n "${NCC_WINDOWS_PFX_PATH:-}" ] && command -v osslsigncode >/dev/null 2>&1 && in_csv "$OS_LIST" windows; then
    local f
    shopt -s nullglob
    for f in dist/ncc-orchestrator-windows-*.exe dist/ncc-api-server-windows-*.exe dist/ncc-ui-server-windows-*.exe; do
      osslsigncode sign -pkcs12 "$NCC_WINDOWS_PFX_PATH" -pass "${NCC_WINDOWS_PFX_PASSWORD:-}" \
        -h sha256 -t "${NCC_WINDOWS_TIMESTAMP_URL:-http://timestamp.digicert.com}" \
        -n "NCC Orchestrator (open-source project)" \
        -i "https://github.com/lTSPV75BRO/Nutanix-ncc-orchestrator" \
        -in "$f" -out "$f.signed"
      mv "$f.signed" "$f"
    done
    shopt -u nullglob
  fi
  if [ -n "${NCC_LINUX_GPG_KEY_ID:-}" ] && command -v gpg >/dev/null 2>&1 && in_csv "$OS_LIST" linux; then
    local f
    shopt -s nullglob
    for f in dist/ncc-orchestrator-linux-*; do
      gpg --batch --yes --local-user "$NCC_LINUX_GPG_KEY_ID" --detach-sign --armor --output "$f.asc" "$f"
    done
    shopt -u nullglob
  fi
}
sign_binaries

build_images() {
  local arch stage primary name file tag_arch tag_version tag_latest
  if [ -z "$ENGINE" ]; then
    resolve_container_engine
  fi
  if [ ! -d dist/frontend-dist ]; then
    die "dist/frontend-dist is missing; do not combine --skip-frontend with --docker unless that directory already exists"
  fi
  # Podman prompts for short names such as alpine:3.24 when more than one
  # registry is configured. Point those names at docker.io and do not ask.
  if [ "$ENGINE" = "podman" ]; then
    export CONTAINERS_SHORT_NAME_MODE=permissive
  fi
  sync_podman_trust
  stage=dist/.image-context
  rm -rf "$stage"
  mkdir -p "$stage"
  primary="${DOCKER_ARCH%%,*}"
  for arch in $(echo "$DOCKER_ARCH" | tr ',' ' '); do
    for name in ncc-orchestrator ncc-api-server ncc-ui-server; do
      if [ ! -f "dist/${name}-linux-${arch}" ]; then
        die "missing dist/${name}-linux-${arch}; build it with --os linux --arch ${arch}"
      fi
      cp "dist/${name}-linux-${arch}" "$stage/"
    done
  done
  cp -R dist/frontend-dist "$stage/frontend-dist"
  collect_host_cas "$stage/ncc-host-cas.pem"
  for arch in $(echo "$DOCKER_ARCH" | tr ',' ' '); do
    for spec in "deploy/docker/Dockerfile.runner|nutanix-ncc-orchestrator" "deploy/docker/Dockerfile.api|ncc-api-server" "deploy/docker/Dockerfile.ui|ncc-ui-server"; do
      file="${spec%%|*}"
      name="${spec##*|}"
      tag_arch="${REGISTRY}/${IMAGE_PREFIX}/${name}:${VERSION}-${arch}"
      tag_version="${REGISTRY}/${IMAGE_PREFIX}/${name}:${VERSION}"
      echo "${ENGINE} build ${name} linux/${arch}"
      fmt=""
      if [ "$ENGINE" = "podman" ]; then
        # OCI images drop HEALTHCHECK. Docker format keeps it, matching docker build.
        fmt="--format docker"
      fi
      # shellcheck disable=SC2086
      "$ENGINE" build $fmt \
        --platform "linux/${arch}" \
        --build-arg "TARGETARCH=${arch}" \
        --build-arg "VERSION=${VERSION}" \
        --build-arg "STREAM=${STREAM}" \
        --build-arg "GIT_REVISION=${GIT_REVISION}" \
        --build-arg "BUILD_DATE=${BUILD_DATE}" \
        -f "$file" \
        -t "$tag_arch" \
        "$stage"
      if [ "$arch" = "$primary" ]; then
        "$ENGINE" tag "$tag_arch" "$tag_version"
        if [ "$DO_LATEST" = 1 ] && [ "$STREAM" = "Release" ]; then
          tag_latest="${REGISTRY}/${IMAGE_PREFIX}/${name}:latest"
          "$ENGINE" tag "$tag_arch" "$tag_latest"
        fi
      fi
      if [ "$DO_PUSH" = 1 ]; then
        "$ENGINE" push "$tag_arch"
        if [ "$arch" = "$primary" ]; then
          "$ENGINE" push "$tag_version"
          if [ "$DO_LATEST" = 1 ] && [ "$STREAM" = "Release" ]; then
            "$ENGINE" push "${REGISTRY}/${IMAGE_PREFIX}/${name}:latest"
          fi
        fi
      fi
    done
  done
  echo "DOCKER_OK engine=${ENGINE} registry=${REGISTRY} prefix=${IMAGE_PREFIX} version=${VERSION} stream=${STREAM}"
}

# Public certificates from the host trust store. Used so a Podman machine can
# pull base images through TLS inspection, and so apk can reach its mirror
# during the image build. Private keys are never exported.
collect_host_cas() {
  local dest="$1"
  : > "$dest"
  if [ -n "${NCC_CONTAINER_CA:-}" ] && [ -f "$NCC_CONTAINER_CA" ]; then
    cat "$NCC_CONTAINER_CA" >> "$dest"
  fi
  if [ "$HOST_KERNEL" = "darwin" ] && command -v security >/dev/null 2>&1; then
    security find-certificate -a -p /Library/Keychains/System.keychain >> "$dest" 2>/dev/null || true
    if [ -f "${HOME}/Library/Keychains/login.keychain-db" ]; then
      security find-certificate -a -p "${HOME}/Library/Keychains/login.keychain-db" >> "$dest" 2>/dev/null || true
    fi
  elif [ "$HOST_KERNEL" = "linux" ]; then
    if [ -d /usr/local/share/ca-certificates ]; then
      cat /usr/local/share/ca-certificates/*.crt >> "$dest" 2>/dev/null || true
    fi
    if [ -d /etc/pki/ca-trust/source/anchors ]; then
      cat /etc/pki/ca-trust/source/anchors/*.crt /etc/pki/ca-trust/source/anchors/*.pem >> "$dest" 2>/dev/null || true
    fi
  fi
}

sync_podman_trust() {
  local pem remote sum current
  [ "$ENGINE" = "podman" ] || return 0
  remote=$("$ENGINE" info --format '{{.Host.ServiceIsRemote}}' 2>/dev/null || echo false)
  [ "$remote" = "true" ] || return 0
  pem=$(mktemp)
  collect_host_cas "$pem"
  if ! grep -q 'BEGIN CERTIFICATE' "$pem"; then
    rm -f "$pem"
    return 0
  fi
  if ! "$ENGINE" machine ssh -- true >/dev/null 2>&1; then
    echo "podman: machine ssh is unavailable; registry pulls use the machine trust store as it is" >&2
    rm -f "$pem"
    return 0
  fi
  sum=$(sha256_file "$pem")
  current=$("$ENGINE" machine ssh -- cat /etc/pki/ca-trust/source/anchors/ncc-host-cas.sha 2>/dev/null || true)
  current=$(printf '%s' "$current" | tr -d '[:space:]')
  if [ "$current" = "$sum" ]; then
    rm -f "$pem"
    echo "podman: host CA trust already installed in the machine"
    return 0
  fi
  echo "podman: installing host CA certificates into the machine trust store"
  "$ENGINE" machine ssh -- sudo tee /etc/pki/ca-trust/source/anchors/ncc-host-cas.pem >/dev/null < "$pem"
  printf '%s\n' "$sum" | "$ENGINE" machine ssh -- sudo tee /etc/pki/ca-trust/source/anchors/ncc-host-cas.sha >/dev/null
  if ! "$ENGINE" machine ssh -- sudo sh -c 'update-ca-trust extract >/dev/null 2>&1 || update-ca-certificates >/dev/null 2>&1'; then
    echo "podman: could not refresh the machine trust store" >&2
  fi
  rm -f "$pem"
}

# For Linux x86_64
GOOS=linux GOARCH=amd64 go build -ldflags "-w -s \
  -X main.Version=1.1.0_LINUX_AMD64 \
  -X main.BuildDate=$(date -u '+%Y-%m-%dT%H:%M:%SZ') \
  -X main.Stream=Prod \
  -X main.GoVersion=$(go version | cut -d ' ' -f 3)" \
  -o ncc-orchestrator-linux-amd64
